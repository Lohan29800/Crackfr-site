const DEFAULT_AUDIO_URL = "https://stream.zeno.fm/f3wvbb7538quv"; // Charger les réglages depuis l'API Vercel (Neon) au démarrage document.addEventListener("DOMContentLoaded", async () => { await loadSettingsFromDB(); checkAccessState(); renderLogs(); }); // Fonction globale pour récupérer les réglages depuis l'API async function loadSettingsFromDB() { try { const response = await fetch('/api/settings'); const data = await response.json(); // On stocke temporairement dans window pour les utiliser facilement window.siteSettings = { status: data.maint_status || 'normal', reason: data.maint_reason || 'Mise à jour majeure du portail et des serveurs de jeu en cours.', banner: data.maint_banner || '', audioUrl: data.audio_url || DEFAULT_AUDIO_URL, // L'authentification staff reste en session locale pour des raisons pratiques auth: localStorage.getItem('auth') === '1' }; } catch (err) { console.error("Erreur de chargement depuis la BD, fallback local :", err); window.siteSettings = { status: 'normal', reason: 'Mise à jour majeure du portail et des serveurs de jeu en cours.', banner: '', audioUrl: DEFAULT_AUDIO_URL, auth: localStorage.getItem('auth') === '1' }; } loadSettingsToUI(); } function getSettings() { return window.siteSettings || { status: 'normal', reason: 'Mise à jour majeure du portail et des serveurs de jeu en cours.', banner: '', audioUrl: DEFAULT_AUDIO_URL, auth: localStorage.getItem('auth') === '1' }; } function loadSettingsToUI() { const config = getSettings(); const bannerEl = document.getElementById('global-banner'); const bannerTextEl = document.getElementById('banner-text'); if(config.banner && config.banner.trim() !== '') { bannerTextEl.innerText = config.banner; bannerEl.classList.remove('hidden'); } else { bannerEl.classList.add('hidden'); } const audioElem = document.getElementById('main-audio-stream'); audioElem.src = config.audioUrl; document.getElementById('admin-maint-status').value = config.status; document.getElementById('admin-maint-reason').value = config.reason; document.getElementById('admin-banner-text').value = config.banner; document.getElementById('admin-audio-url').value = config.audioUrl; document.getElementById('display-maint-reason').innerText = config.reason; } function checkAccessState() { const config = getSettings(); const maintScreen = document.getElementById('maintenance-screen'); const adminDash = document.getElementById('admin-dashboard'); const mainContent = document.querySelector('main'); if (config.status === 'maintenance' && !config.auth) { maintScreen.classList.remove('hidden'); adminDash.classList.add('hidden'); } else if (config.auth) { maintScreen.classList.add('hidden'); adminDash.classList.remove('hidden'); mainContent.classList.add('hidden'); } else { maintScreen.classList.add('hidden'); adminDash.classList.add('hidden'); mainContent.classList.remove('hidden'); } } // Sauvegarder les réglages directement dans Neon via l'API async function saveAdminSettings() { const status = document.getElementById('admin-maint-status').value; const reason = document.getElementById('admin-maint-reason').value; const banner = document.getElementById('admin-banner-text').value; const audioUrl = document.getElementById('admin-audio-url').value; try { const response = await fetch('/api/settings', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ maint_status: status, maint_reason: reason, maint_banner: banner, audio_url: audioUrl }) }); if (response.ok) { await loadSettingsFromDB(); checkAccessState(); addLog("Mise à jour des réglages du portail par le Staff (enregistré sur Neon)."); showToast("Réglages sauvegardés sur la base de données ! ⚡"); } else { showToast("Erreur lors de la sauvegarde ❌"); } } catch (err) { console.error("Erreur réseau :", err); showToast("Erreur de connexion au serveur ⚠️"); } }
